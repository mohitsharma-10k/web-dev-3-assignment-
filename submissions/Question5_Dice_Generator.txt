// File: dice.js
const crypto = require('crypto');

// Function to roll a 6-sided dice using crypto module
function rollDice() {
    // crypto.randomInt(min, max) generates a cryptographically secure random integer
    // where min is inclusive (1) and max is exclusive (7) -> produces 1, 2, 3, 4, 5, or 6
    const result = crypto.randomInt(1, 7); 
    console.log(`Dice Rolled: ${result}`);
}

// Simulate multiple dice rolls using a loop
console.log("Rolling dice 5 times using loops:");
for (let i = 1; i <= 5; i++) {
    rollDice();
}
